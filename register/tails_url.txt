const targets = raw
  .trim()
  .split('\n')
  .map(line => {
    const [sourceTitle, url] = line.split(/\t+/);
    return {
      sourceTitle: sourceTitle?.trim(),
      url: url?.trim()
    };
  })
  .filter(x => x.url);

console.log(`수집 대상: ${targets.length}개`);

const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

async function collectTails() {
  const results = [];

  for (let i = 0; i < targets.length; i++) {
    const { sourceTitle, url } = targets[i];

    console.log(`[${i + 1}/${targets.length}] ${sourceTitle}`);

    try {
      const html = await fetch(url).then(r => {
        if (!r.ok) throw new Error(`HTTP ${r.status}`);
        return r.text();
      });

      const doc = new DOMParser().parseFromString(html, 'text/html');

      // 본문 section 후보 중 "테일스 + 주식"이 들어있는 영역 탐색
      const sections = [...doc.querySelectorAll('section')];

      let section = sections.find(s => {
        const t = s.innerText || s.textContent || '';
        return t.includes('테일스') && t.includes('주식:');
      });

      // section 구조가 다른 글 대비 fallback
      if (!section) {
        const candidates = [...doc.querySelectorAll('article, main, div')];

        section = candidates
          .filter(el => {
            const t = el.innerText || el.textContent || '';
            return t.includes('테일스') && t.includes('주식:') && t.includes('능력');
          })
          .sort((a, b) =>
            (a.innerText || a.textContent || '').length -
            (b.innerText || b.textContent || '').length
          )[0];
      }

      if (!section) {
        throw new Error('본문 영역 없음');
      }

      let text = (section.innerText || section.textContent || '')
        .replace(/\r/g, '')
        .replace(/[ \t]+\n/g, '\n')
        .replace(/\n[ \t]+/g, '\n')
        .replace(/\n{3,}/g, '\n\n')
        .trim();

      // 페이지 앞뒤 잡텍스트가 있으면 실제 제목부터 잘라냄
      const titleMatch = text.match(/\d+\s*\.\s*.+?의 테일스/);

      if (titleMatch && titleMatch.index > 0) {
        text = text.slice(titleMatch.index).trim();
      }

      const header =
        text.match(
          /(\d+)\s*\.\s*(.+?)의 테일스\s*\(\s*파트너\s*:\s*(.+?)(?:\s*님)?\s*\)/
        );

      if (!header) {
        throw new Error('번호/이름/파트너 파싱 실패');
      }

      const number = header[1].trim();
      const name = header[2].trim();

      let owner = header[3].trim();
      owner = owner.replace(/\s*님\s*$/, '').trim();

      results.push({
        number,
        name,
        owner,
        text,
        url
      });

    } catch (e) {
      console.error(`실패: ${sourceTitle}`, e);

      results.push({
        sourceTitle,
        url,
        error: e.message
      });
    }

    // Wix에 너무 빠르게 요청하지 않도록 약간 쉬기
    await sleep(300);
  }

  console.log('====================');
  console.log('완료!', results);
  console.log(
    '정상:',
    results.filter(x => !x.error).length,
    '실패:',
    results.filter(x => x.error).length
  );
  console.log('====================');

  window.tailsRaw = results;

  return results;
}
